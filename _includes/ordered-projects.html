{%- comment -%} Builds `ordered_projects`: every project, grouped by category (in the order of _data/categories.yml), then by `order` (if set), then newest first. This single ordering drives the work index, the sidebar and prev/next links. {%- endcomment -%} {%- assign ordered_projects = "" | split: "" -%} {%- for cat in site.data.categories -%} {%- assign in_cat = site.projects | where: "category", cat.id | sort: "date" | reverse -%} {%- assign with_order = in_cat | where_exp: "p", "p.order" | sort: "order" -%} {%- assign without_order = in_cat | where_exp: "p", "p.order == nil" -%} {%- assign ordered_projects = ordered_projects | concat: with_order | concat: without_order -%} {%- endfor -%}
